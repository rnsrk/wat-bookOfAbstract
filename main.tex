% Wichtig: \fancypagestyle{fancy}{...} (statt nur \pagestyle{fancy} + losen

  \fancyhf{}%
  \fancyhead[L]{\sffamily\small\currentcategory}%

  \renewcommand{\footrulewidth}{0pt}%

% \category{Name}: leitet eine neue Kategorie ein (Trennblatt + Kolumnentitel)
\newcommand{\category}[1]{%
  \clearpage
  \thispagestyle{kategorie}%
  \renewcommand{\currentcategory}{#1}%
  \markright{}%
  \phantomsection
  \addcontentsline{toc}{part}{#1}%
  \null\vspace*{0.25\textheight}%
  \begin{center}
    {\Huge\sffamily\bfseries\MakeTextUppercase{#1}}\par
  \end{center}
  \clearpage
}

% \backmattersection{Name}: top-level Eintrag ohne Trennblatt (für Backmatter)
\newcommand{\backmattersection}[1]{%
  \clearpage
  \renewcommand{\currentcategory}{#1}%
  \markright{}%
  \phantomsection
  \addcontentsline{toc}{part}{#1}%
  \section*{#1}%
}

% beitrag-Umgebung: kapselt einen einzelnen Abstract.
%   [#1] = optional: Autor*innennamen als Klartext für das Inhaltsverzeichnis
%          (z. B. [Anke Albrecht, Bernd Braun]). Fehlt das Argument, entfällt
%          nur die Autorenzeile im TOC -- die Datei baut weiterhin fehlerfrei.
%   #2   = Titel
%   #3   = Autor*innenblock für die Anzeige (Name, ORCID, E-Mail, Affiliation -- mit \\)
% - Resettet Zähler (section, subsection, ..., figure, table) auf 0.
% - Setzt \rightmark auf den Titel (Kopfzeile zeigt Kategorie | Titel).
% - Trägt Titel und Autorenzeile manuell ins Inhaltsverzeichnis ein und
%   neutralisiert dann \addcontentsline, damit interne \section{...} usw.
%   das globale TOC nicht verschmutzen.
% - Öffnet eine refsection; am Ende wird die abstract-eigene
%   Bibliografie ausgegeben.
\newenvironment{beitrag}[3][]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{section}{#2}%
  \ifblank{#1}{}{%
    \addtocontents{toc}{\protect\nopagebreak}%
    \addcontentsline{toc}{subsection}{#1}%
  }%
  \markright{#2}%
  \setcounter{section}{0}%
  \setcounter{subsection}{0}%
  \setcounter{subsubsection}{0}%
  \setcounter{figure}{0}%
  \setcounter{table}{0}%
  \setcounter{footnote}{0}%
  \begingroup
    \centering
    {\Large\sffamily\bfseries #2\par}%
    \medskip
    {\normalsize #3\par}%
  \endgroup
  \vspace{1.5em}
  \let\savedaddcontentsline\addcontentsline
  \renewcommand{\addcontentsline}[3]{}%
  \begin{refsection}%
}{%
  \printbibliography[heading=abstractbib]%
  \end{refsection}%
  \let\addcontentsline\savedaddcontentsline%
}
